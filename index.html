Clean, Idiomatic APIs

No convoluted DSLs or complex boilerplate. Just clean Dart functions that integrate seamlessly with your existing test runners. 

import 'package:test/test.dart';
import 'package:hegeltest/hegeltest.dart';

void main() {
  hegelTest('payment handles currency matrix & zero amounts', (tc) {
    // Weighted domain generator
    final currency = tc.draw(sampledWeighted([(80, 'USD'), (15, 'EUR'), (5, 'JPY')]));
    final amount = tc.draw(integers(min: 0, max: 500000), label: 'amount');

    // Classify distributions conditionally
    tc.classify(amount == 0, 'zero_amount');
    tc.classify(amount > 250000, 'large_amount');

    // Coverage Contract: assert zero-amount edge cases tested >= 10%
    tc.cover(10.0, amount == 0, 'zero_amount_coverage');

    final invoice = createInvoice(amount: amount, currency: currency);
    expect(invoice.total, equals(amount));
  });
}

    await tester.pumpWidget(
      MaterialApp(
        home: Center(
          child: Padding(
            padding: EdgeInsets.symmetric(horizontal: paddingHorizontal, vertical: paddingVertical),
            child: Text(label),
          ),
        ),
      ),
    );

    // Verify widget rendered without throwing layout overflow assertions
    expect(find.byType(Text), findsOneWidget);
  });
}

import 'package:flutter_test/flutter_test.dart';
import 'package:hegeltest_flutter/hegeltest_flutter.dart';

class BoundedQueueMachine extends StateMachine {
  final queue = <int>[];

  @override
  List<StateRule> get rules => [
    StateRule('enqueue', execute: (tc) {
      final item = tc.draw(integers());
      queue.add(item);
    }),
    StateRule('dequeue', precondition: () => queue.isNotEmpty, execute: (tc) {
      queue.removeAt(0);
    }),
  ];

  @override
  List<StateInvariant> get invariants => [
    StateInvariant('queue size is non-negative', check: (tc) {
      expect(queue.length, greaterThanOrEqualTo(0));
    }),
  ];
}

void main() {
  hegelFlutterStatefulTest('queue satisfies invariants', () => BoundedQueueMachine());
}

// FOUNDATIONAL DESIGN

Engineered from the Metal Up

How LetsTestTools harnesses native fuzzing without burdening developers with Rust build toolchains.

01

Dart Native Assets (Hook Engine)

Using Dart's modern hook/build.dart build hook, precompiled native libraries (libhegel_c) are automatically verified via SHA-256 sidecars and packaged into your test artifact. Contributors never need cargo or a C toolchain installed.

02

Zero-Allocation FFI Interop

Execution data, mutations, and seed buffers are passed across Dart FFI via high-speed native pointers with pinned memory. Generators achieve millions of draws per second with negligible garbage collection overhead.

03

Deterministic Replay Database

Every discovered counterexample is hashed and persisted to .hegel/examples/. An automated .gitignore prevents repository pollution. When your test runs again locally or in CI, Phase.reuse executes the failure immediately.
